def get_bounding_box(self, anomaly_map: np.ndarray, original_size: tuple) -> tuple:
    """
    Returns (x, y, w, h) bounding box around the highest scoring patch.
    """
    H, W = original_size
    amap = gaussian_filter(anomaly_map.astype(np.float32), sigma=2)
    amap_norm = self._normalise(amap)
    amap_resized = cv2.resize(amap_norm, (W, H), interpolation=cv2.INTER_LINEAR)

    # Find the single highest scoring location
    max_loc = np.unravel_index(amap_resized.argmax(), amap_resized.shape)
    y_center, x_center = max_loc

    # Draw a fixed-size box around it
    box_size = min(H, W) // 6
    x = max(0, x_center - box_size // 2)
    y = max(0, y_center - box_size // 2)
    w = min(W - x, box_size)
    h = min(H - y, box_size)

    return (x, y, w, h)
