import json
from pathlib import Path

for cat in ['engine_wiring', 'pipe_clips', 'pipe_staples']:
    metrics_file = Path(f'outputs/{cat}/metrics.json')
    if metrics_file.exists():
        m = json.load(open(metrics_file))
        print(f"\n{cat}")
        print(f"  Optimal threshold : {m.get('img_threshold', 'N/A')}")
        print(f"  Image AUROC       : {m.get('img_auroc', 'N/A')}")
        print(f"  Image F1          : {m.get('img_f1', 'N/A')}")
